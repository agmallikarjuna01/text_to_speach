async function processData() { const name = document.getElementById("name").value; const file = document.getElementById("file").files[0]; const formData = new FormData(); if (file) { formData.append("file", file); } else { formData.append("name", name); } const response = await fetch("http://127.0.0.1:5000/api/convert", { method: "POST", body: formData }); if (!response.ok) { const error = await response.json(); document.getElementById("result").innerText = error.error || "Unable to create audio."; return; } const audioBlob = await response.blob(); const audio = document.getElementById("audio"); audio.src = URL.createObjectURL(audioBlob); audio.hidden = false; await audio.play(); }
